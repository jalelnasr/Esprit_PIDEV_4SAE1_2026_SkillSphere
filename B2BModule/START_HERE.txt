🚀 DÉMARRER EN 5 MINUTES
═════════════════════════════════════════════════════════════════════

# 1. Compiler
mvn clean compile

# 2. Lancer
mvn spring-boot:run

# 3. Tester (cURL)
curl -X POST http://localhost:8083/api/b2b/applications/notify \
  -H "Content-Type: application/json" \
  -d '{
    "candidateEmail": "test@gmail.com",
    "candidateName": "John Doe",
    "jobTitle": "Developer",
    "companyName": "TechCorp",
    "status": "ACCEPTED"
  }'

# 4. Email reçu! ✅
